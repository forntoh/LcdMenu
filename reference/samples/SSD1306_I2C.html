231void clearInputValues() {
232    ItemInput* user = static_cast<ItemInput*>(inputItems[2]);
233    ItemInput* noteInput = static_cast<ItemInput*>(inputItems[3]);
234    ItemInput* tagInput = static_cast<ItemInput*>(inputItems[4]);
235
236    char* oldUser = user->getValue();
237    char* oldNote = noteInput->getValue();
238    char* oldTag = tagInput->getValue();
239
240    user->setValue(cloneText(""));
241    noteInput->setValue(cloneText(""));
242    tagInput->setValue(cloneText(""));
243
244    delete[] oldUser;
245    delete[] oldNote;
246    delete[] oldTag;
247
248    menu.refresh();
249}

342void addDynamicItem() {
343    char text[24];
344    snprintf(text, sizeof(text), "Added #%u", dynamicCounter++);
345    dynamicScreen->addItem(ITEM_BASIC(cloneText(text)));
346    menu.refresh();

349void insertDynamicItem() {
350    char text[24];
351    snprintf(text, sizeof(text), "Inserted #%u", dynamicCounter++);
352    dynamicScreen->addItemAt(dynamicBaseCount, ITEM_BASIC(cloneText(text)));
353    menu.refresh();
354}

356void removeDynamicAtHead() {
357    if (dynamicScreen->size() <= dynamicBaseCount) {
358        return;
359    }
360    dynamicScreen->removeItemAt(dynamicBaseCount);
361    menu.refresh();
362}
363
364void removeDynamicLast() {
365    if (dynamicScreen->size() <= dynamicBaseCount) {
366        return;
367    }
368    dynamicScreen->removeLastItem();
369    menu.refresh();
370}
371
372void clearDynamicItems() {
373    while (dynamicScreen->size() > dynamicBaseCount) {
374        dynamicScreen->removeLastItem();
375    }
376    menu.refresh();
377}

379void updateTelemetry() {
380    static unsigned long lastTick = 0;
381    static int8_t temperatureDirection = 1;
382
383    unsigned long now = millis();
384    if (now - lastTick < 1000) {
385        return;
386    }
387    lastTick = now;
388
389    uptimeSeconds++;
390    temperatureC += 0.2f * temperatureDirection;
391    if (temperatureC > 28.0f || temperatureC < 21.0f) {
392        temperatureDirection = -temperatureDirection;
393    }
394    voltage = 3.20f + 0.01f * (uptimeSeconds % 20);
395    rpm = 1100 + (uptimeSeconds % 16) * 40;
396}
